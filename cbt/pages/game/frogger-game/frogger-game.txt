### **Explanation of Each File and Folder**

#### **Root Folder: `frogger-game/`**

*   This is the main folder that contains your entire project. You can name it whatever you like (e.g., `educational-game`, `my-frogger`).

#### **Core Files (in the root folder)**

*   **`📄 index.html`**
    *   **Purpose:** This is the main entry point of your website. It holds the structure of your game, including the `<canvas>` element where the game is drawn and any surrounding HTML like titles or buttons. It links to your CSS and JavaScript files.
    *   **Role:** The "skeleton" of your project.

*   **`📄 style.css`**
    *   **Purpose:** This file contains all the styling rules. It controls the colors, layout, fonts, and overall visual appearance of your `index.html` page, including the game container and any on-screen messages.
    *   **Role:** The "skin" or "visual design" of your project.

*   **`📄 script.js`**
    *   **Purpose:** This is where all the game's logic resides. It handles player movement, drawing on the canvas, checking for collisions, loading level data, and managing the game state (playing, win, lose).
    *   **Role:** The "brain" of your project.

*   **`📄 levels.json`**
    *   **Purpose:** This is a pure data file. It describes the configuration of your game levels (lane types, obstacle speeds, etc.) in a structured, easy-to-read format. Your `script.js` reads from this file to build the game world.
    *   **Role:** The "blueprint" or "configuration data" for your game.

#### **The `assets/` Folder (Optional but Recommended)**

*   **Purpose:** This folder is a standard convention for storing all static media files that your game uses. While our current simple version doesn't use external images or sounds, any real game would. Organizing them here keeps your root directory clean.
*   **Role:** The "media library" for your project.

*   **`📁 images/` (sub-folder)**
    *   This is where you would store all your image files, such as sprites for the player (`frog.png`), vehicles (`car.png`), or background elements.

*   **`📁 sounds/` (sub-folder)**
    *   This is where you would keep all your audio files, like sound effects for hopping (`hop-sound.mp3`) or background music.

This structure is scalable, easy to understand, and follows best practices used by web developers everywhere.








JAVA

document.addEventListener('DOMContentLoaded', () => {

    // --- DOM ELEMENTS & CONTEXT ---
    const canvas = document.getElementById('gameCanvas');
    const ctx = canvas.getContext('2d');
    const messageEl = document.getElementById('game-message');
    const overlayEl = document.getElementById('game-overlay');

    // --- GAME CONFIG & STATE ---
    const TILE_SIZE = 50;
    canvas.width = 9 * TILE_SIZE; // 9 tiles wide
    canvas.height = 9 * TILE_SIZE; // 9 tiles high

    let gameState = 'START'; // START, PLAYING, WIN, LOSE
    let levelData = null; // Will hold our JSON data
    let lanes = [];

    const player = {
        x: 4 * TILE_SIZE, // Start in the middle
        y: 8 * TILE_SIZE, // Start at the bottom
        size: TILE_SIZE * 0.8,
        speed: TILE_SIZE,
        color: '#2ecc71' // A froggy green
    };

    /**
     * Fetches the level data from the JSON file.
     * This is an asynchronous function, a key modern JS concept.
     */
    async function loadLevelData() {
        try {
            const response = await fetch('levels.json');
            if (!response.ok) {
                throw new Error(`HTTP error! status: ${response.status}`);
            }
            levelData = await response.json();
            console.log("Level data loaded successfully:", levelData);
        } catch (error) {
            console.error("Could not load level data:", error);
            messageEl.textContent = "Error Loading Level!";
        }
    }

    /**
     * Sets up the game based on the loaded level data.
     */
    function setupLevel() {
        // Reset player to start position
        player.x = 4 * TILE_SIZE;
        player.y = 8 * TILE_SIZE;

        const currentLevel = levelData.level1;
        lanes = currentLevel.lanes.map((laneConfig, index) => {
            return {
                ...laneConfig,
                y: index * TILE_SIZE,
                obstacles: spawnObstacles(laneConfig)
            };
        });
    }

    /**
     * Creates the initial set of obstacles for a given lane.
     */
    function spawnObstacles(laneConfig) {
        if (laneConfig.type === 'safe') return [];

        const obstacles = [];
        let currentX = Math.random() * canvas.width; // Start at a random position

        for (const obsConfig of laneConfig.obstacles) {
            obstacles.push({
                x: currentX,
                width: obsConfig.width * TILE_SIZE,
                color: laneConfig.type === 'road' ? '#c0392b' : '#8e44ad' // Car or Log color
            });
            // Space out the next obstacle
            currentX += obsConfig.width * TILE_SIZE + (2 + Math.random() * 3) * TILE_SIZE;
        }
        return obstacles;
    }


    // --- GAME LOOP & DRAWING ---

    function gameLoop() {
        if (gameState === 'PLAYING') {
            update();
            draw();
        }
        requestAnimationFrame(gameLoop);
    }

    function update() {
        // Move obstacles
        lanes.forEach(lane => {
            if (lane.speed) {
                lane.obstacles.forEach(obs => {
                    obs.x += lane.speed * lane.direction;

                    // Wrap obstacles around the screen
                    if (lane.direction === 1 && obs.x > canvas.width) {
                        obs.x = -obs.width;
                    } else if (lane.direction === -1 && obs.x < -obs.width) {
                        obs.x = canvas.width;
                    }
                });
            }
        });

        // Check for collisions
        checkCollisions();
    }

    function draw() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);

        // Draw lanes and obstacles
        lanes.forEach(lane => {
            ctx.fillStyle = lane.color;
            ctx.fillRect(0, lane.y, canvas.width, TILE_SIZE);
            lane.obstacles.forEach(obs => {
                ctx.fillStyle = obs.color;
                ctx.fillRect(obs.x, lane.y + TILE_SIZE * 0.1, obs.width, TILE_SIZE * 0.8);
            });
        });

        // Draw player
        ctx.fillStyle = player.color;
        const playerMargin = (TILE_SIZE - player.size) / 2;
        ctx.fillRect(player.x + playerMargin, player.y + playerMargin, player.size, player.size);
    }

    // --- COLLISION & GAME STATE LOGIC ---

    function checkCollisions() {
        const laneIndex = Math.floor(player.y / TILE_SIZE);
        const currentLane = lanes[laneIndex];

        if (currentLane.type === 'road') {
            for (const obs of currentLane.obstacles) {
                if (player.x < obs.x + obs.width && player.x + TILE_SIZE > obs.x) {
                    // Collision with a car!
                    resetGame("You got hit!");
                    return;
                }
            }
        } else if (currentLane.type === 'river') {
            let onLog = false;
            for (const obs of currentLane.obstacles) {
                if (player.x < obs.x + obs.width && player.x + TILE_SIZE > obs.x) {
                    onLog = true;
                    // Move player with the log
                    player.x += currentLane.speed * currentLane.direction;
                    break;
                }
            }
            if (!onLog) {
                // Fell in the water!
                resetGame("You fell in!");
                return;
            }
        }
        // Check for win condition
        if (laneIndex === 0) {
            winGame();
        }
    }
    
    function resetGame(message) {
        gameState = 'LOSE';
        messageEl.textContent = message;
        overlayEl.style.display = 'flex';
        // After a delay, allow restart
        setTimeout(() => {
            gameState = 'START';
            messageEl.textContent = 'Press Enter to Restart';
        }, 1500);
    }
    
    function winGame() {
        gameState = 'WIN';
        messageEl.textContent = 'You Made It!';
        overlayEl.style.display = 'flex';
        // After a delay, allow restart
        setTimeout(() => {
            gameState = 'START';
            messageEl.textContent = 'Press Enter to Play Again';
        }, 1500);
    }


    // --- EVENT LISTENERS ---

    window.addEventListener('keydown', (e) => {
        if (gameState === 'PLAYING') {
            e.preventDefault(); // Prevent page from scrolling
            switch (e.key) {
                case 'ArrowUp':
                    if (player.y > 0) player.y -= player.speed;
                    break;
                case 'ArrowDown':
                    if (player.y < canvas.height - TILE_SIZE) player.y += player.speed;
                    break;
                case 'ArrowLeft':
                    if (player.x > 0) player.x -= player.speed;
                    break;
                case 'ArrowRight':
                    if (player.x < canvas.width - TILE_SIZE) player.x += player.speed;
                    break;
            }
        } else if (gameState === 'START' && e.key === 'Enter') {
            setupLevel(); // Setup the level from JSON data
            gameState = 'PLAYING';
            overlayEl.style.display = 'none';
        }
    });

    // --- INITIALIZE GAME ---
    loadLevelData().then(() => {
        // This ensures the game loop only starts after data is loaded
        requestAnimationFrame(gameLoop);
    });
});
