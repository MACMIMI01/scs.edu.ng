Excellent question! Yes, absolutely. The tech stack you've listed (HTML, JS, JSON, CSS) and the use of a static HTTP file server is the perfect combination for creating a wide variety of classic web games.

Let's build a Memory Match Game (also known as Concentration). This is a fantastic educational project because it perfectly demonstrates all the technologies working together:

JSON: To define the cards, separating the game's data from its logic.

JavaScript: To fetch and shuffle the JSON data, handle the game logic (flipping, matching, winning), and manipulate the DOM.

CSS: To create the card styles, the 3D "flip" animation, and the overall look and feel.

HTML: To provide the basic structure and container for the game.

Static HTTP Server: This is crucial. Your script.js will use the fetch() API to load cards.json. For security reasons, modern browsers block fetch requests if you just open the index.html file from your local filesystem (the file:/// protocol). Running a simple server (like VS Code's "Live Server" extension or Python's http.server) serves the files over the http:// protocol, allowing fetch to work correctly.

Here is the complete code to create a fully functional and polished Memory Match game.

How to Set Up the Project

Create a new folder named memory-game/.

Inside this folder, create the three files below: cards.json, index.html, and style.css.

Copy and paste the code into the corresponding files.

Important: Open this folder in a code editor like VS Code and use the "Live Server" extension (or any other static server) to run it.

4. The Logic: script.js

This is where the entire game comes to life. It's heavily commented to explain each step.

Generated javascript

document.addEventListener('DOMContentLoaded', () => { // --- DOM ELEMENTS --- const gameBoard = document.querySelector('.game-board'); const movesCountEl = document.getElementById('moves-count'); const timerDisplayEl = document.getElementById('timer-display'); const winOverlay = document.getElementById('win-overlay'); const restartBtn = document.getElementById('restart-btn'); // --- GAME STATE VARIABLES --- let cards = []; let flippedCards = []; let matchedPairs = 0; let moves = 0; let timer = 0; let timerInterval = null; let isLocked = false; // Prevents clicking during the "unflip" delay /** * Fetches card data from JSON and initializes the game. */ async function setupGame() { try { const response = await fetch('cards.json'); const cardData = await response.json(); // Create pairs by doubling the card data const gameCards = [...cardData, ...cardData]; // Shuffle the cards shuffle(gameCards); // Create card elements and add to the board createCardElements(gameCards); startTimer(); } catch (error) { console.error("Could not set up the game:", error); gameBoard.innerHTML = "<p>Error: Could not load card data.</p>"; } } /** * Shuffles an array in place using the Fisher-Yates algorithm. */ function shuffle(array) { for (let i = array.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [array[i], array[j]] = [array[j], array[i]]; } } /** * Creates HTML elements for each card and adds event listeners. */ function createCardElements(gameCards) { gameBoard.innerHTML = ''; // Clear board first gameCards.forEach(cardInfo => { const cardElement = document.createElement('div'); cardElement.classList.add('card'); cardElement.dataset.name = cardInfo.name; // Store the card name for matching cardElement.innerHTML = ` <div class="card-face card-front"> <i class="fas fa-question"></i> </div> <div class="card-face card-back"> <i class="fas ${cardInfo.icon}"></i> </div> `; cardElement.addEventListener('click', handleCardClick); gameBoard.appendChild(cardElement); }); } /** * Handles the logic when a card is clicked. */ function handleCardClick(event) { const clickedCard = event.currentTarget; // Ignore clicks if the board is locked, or if the card is already flipped or matched if (isLocked || clickedCard.classList.contains('is-flipped')) { return; } // Flip the card clickedCard.classList.add('is-flipped'); flippedCards.push(clickedCard); // If two cards are flipped, check for a match if (flippedCards.length === 2) { incrementMoves(); isLocked = true; // Lock the board checkForMatch(); } } /** * Checks if the two flipped cards are a match. */ function checkForMatch() { const [cardOne, cardTwo] = flippedCards; const isMatch = cardOne.dataset.name === cardTwo.dataset.name; if (isMatch) { disableMatchedCards(); } else { unflipMismatchedCards(); } } /** * Handles matched cards. */ function disableMatchedCards() { flippedCards.forEach(card => { card.removeEventListener('click', handleCardClick); card.classList.add('is-matched'); }); matchedPairs++; flippedCards = []; isLocked = false; // Unlock the board // Check for win condition if (matchedPairs === 8) { // 8 pairs in our game winGame(); } } /** * Handles mismatched cards by flipping them back after a delay. */ function unflipMismatchedCards() { setTimeout(() => { flippedCards.forEach(card => card.classList.remove('is-flipped')); flippedCards = []; isLocked = false; // Unlock the board }, 1000); // 1-second delay } function winGame() { clearInterval(timerInterval); document.getElementById('final-time').textContent = `${timer}s`; document.getElementById('final-moves').textContent = moves; winOverlay.style.display = 'flex'; } // --- UTILITY FUNCTIONS --- function incrementMoves() { moves++; movesCountEl.textContent = moves; } function startTimer() { timerInterval = setInterval(() => { timer++; timerDisplayEl.textContent = `${timer}s`; }, 1000); } function resetGame() { // Reset all state variables flippedCards = []; matchedPairs = 0; moves = 0; timer = 0; isLocked = false; movesCountEl.textContent = '0'; timerDisplayEl.textContent = '0s'; clearInterval(timerInterval); // Hide win screen and set up a new game winOverlay.style.display = 'none'; setupGame(); } // --- INITIALIZE --- restartBtn.addEventListener('click', resetGame); setupGame(); });
