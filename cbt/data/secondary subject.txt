document.addEventListener('DOMContentLoaded', () => {
  // 1. FIRST ADD ALL YOUR EXAM ARRAYS HERE
  const agriculturalScienceExams = [
    { fileName: '2025-2026-JS1-ACS-1ST', displayName: 'BASIC 7 1ST TERM AGRICULTURAL SCIENCE EXAM 2025/2026' },
    // ... add all other agricultural science exams
  ];
  
  const basicScienceExams = [
    { fileName: '2025-2026-JS1-BSCI-1ST', displayName: 'BASIC 7 1ST TERM BASIC SCIENCE EXAM 2025/2026' },
    // ... add all other basic science exams
  ];
  
  // ... add all your other exam arrays exactly as you have them

  // 2. COMBINE ALL EXAMS INTO ONE ARRAY
  const allExams = [
    ...agriculturalScienceExams,
    ...basicScienceExams,
    ...basicTechnologyExams,
    ...businessStudiesExams,
    ...christianReligiousStudiesExams,
    ...civicEducationExams,
    ...computerStudiesExams,
    ...culturalAndCreativeArtExams,
    ...dictationExams,
    ...juniorDictionExams,
    ...englishStudiesExams,
    ...frenchExams,
    ...historicalStudiesExams,
    ...homeEconomicsExams,
    ...mathematicsExams,
    ...musicExams,
    ...physicalAndHealthEducationExams,
    ...securityEducationExams,
    ...socialStudiesExams
  ];

  // 3. GET DOM ELEMENTS
  const examListElement = document.getElementById('exam-list');
  const searchInput = document.getElementById('search-input');
  const searchButton = document.getElementById('search-button');

  // 4. FUNCTION TO DISPLAY EXAMS
  function displayExams(examsToShow) {
    examListElement.innerHTML = ''; // Clear current list
    
    if (examsToShow.length === 0) {
      examListElement.innerHTML = '<li class="loading">No matching exams found</li>';
      return;
    }
    
    examsToShow.forEach(exam => {
      const listItem = document.createElement('li');
      const anchor = document.createElement('a');
      anchor.href = `test-taking.html?test=${encodeURIComponent(exam.fileName)}`;
      anchor.textContent = exam.displayName;
      listItem.appendChild(anchor);
      examListElement.appendChild(listItem);
    });
  }

  // 5. INITIAL LOAD - SHOW ALL EXAMS
  displayExams(allExams);

  // 6. SEARCH FUNCTIONALITY
  function handleSearch() {
    const searchTerm = searchInput.value.toLowerCase().trim();
    const filteredExams = allExams.filter(exam => 
      exam.displayName.toLowerCase().includes(searchTerm)
    );
    displayExams(filteredExams);
  }

  // 7. EVENT LISTENERS
  searchButton.addEventListener('click', handleSearch);
  searchInput.addEventListener('input', handleSearch);
  searchInput.addEventListener('keypress', (e) => {
    if (e.key === 'Enter') handleSearch();
  });
});
